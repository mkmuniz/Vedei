# Credentials under key names in English, Portuguese and Spanish.
#
# Measured before the localization pass in engine/secrets, a real credential was
# found 100% of the time under an English name, 57% under a Spanish one and 29%
# under a Portuguese one: senha and segredo were never recognized. These cases
# pin the fix. The same value appears under every name, so a difference between
# languages can only come from the name.
#
# Every secret value here is a made-up string shaped like a credential.

# --- English ---
expect generic-password  password = "8f3kq9zLw2Xc7vB1nM4p"
expect generic-password  password: 8f3kq9zLw2Xc7vB1nM4p
expect generic-password  PASSWORD=8f3kq9zLw2Xc7vB1nM4p
expect generic-password  db_password = "8f3kq9zLw2Xc7vB1nM4p"
expect generic-password  db_password: 8f3kq9zLw2Xc7vB1nM4p
expect generic-password  DB_PASSWORD=8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  api_key = "8f3kq9zLw2Xc7vB1nM4p"
expect generic-api-key  api_key: 8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  API_KEY=8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  secret = "8f3kq9zLw2Xc7vB1nM4p"
expect generic-api-key  secret: 8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  SECRET=8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  access_token = "8f3kq9zLw2Xc7vB1nM4p"
expect generic-api-key  access_token: 8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  ACCESS_TOKEN=8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  client_secret = "8f3kq9zLw2Xc7vB1nM4p"
expect generic-api-key  client_secret: 8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  CLIENT_SECRET=8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  auth_key = "8f3kq9zLw2Xc7vB1nM4p"
expect generic-api-key  auth_key: 8f3kq9zLw2Xc7vB1nM4p
expect generic-api-key  AUTH_KEY=8f3kq9zLw2Xc7vB1nM4p

# A reference to a value is not a value. Suppressed in every language,
# including after the key name is translated.
reject  password = "${DB_PASSWORD}"   # environment variable reference
reject  password = os.getenv("DB_PASSWORD")   # read at runtime, no literal
reject  password = "<password>"   # template marker
reject  api_key = "${API_KEY}"   # environment variable reference
reject  the password expired yesterday   # prose, not an assignment
